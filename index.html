/* ===================================== FAST ADAPTIVE CURSOR ===================================== */ let smoothX = null; let smoothY = null; /* * VERY CLOSE FINGER TRACKING * * Small movement → 0.75 * Normal movement → 0.90 * Fast movement → 0.97 * * Higher value = point stays * closer to the real fingertip. */ function smoothPoint(targetX, targetY) { if (smoothX === null) { smoothX = targetX; smoothY = targetY; return; } const dx = targetX - smoothX; const dy = targetY - smoothY; const distance = Math.sqrt( dx * dx + dy * dy ); let adaptiveSmoothing; if (distance < 5) { adaptiveSmoothing = 0.75; } else if (distance < 20) { adaptiveSmoothing = 0.90; } else { adaptiveSmoothing = 0.97; } smoothX = smoothX + dx * adaptiveSmoothing; smoothY = smoothY + dy * adaptiveSmoothing; }
